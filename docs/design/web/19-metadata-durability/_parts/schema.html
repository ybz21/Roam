4. 表设计

单库 <dataDir>/meta.db，schema_meta 管版本，迁移脚本放 cli/ttmux-cli-go/internal/metadb/migrations/。

这一节是 M2 的目标 schema，不是现在的表结构。现在只有 sessions 一张表 按这个方向改到位了（主键换持久 id，加 status / died_at / died_reason / tmux_id / tmux_epoch）； projects 仍是 projects.json（只多了 lastSessionAt 一个字段）， tmux_epochs / project_aliases / archived_at / spawn_cmd / restored_from 都还没有。逐项状态见 §10。

4.1 DDL

CREATE TABLE schema_meta (version INTEGER NOT NULL);

-- tmux server 代次：server pid，随 server 起落而变。
-- 用来把「你手动 kill 了会话」和「机器重启把整个 server 带走了」分开记。
CREATE TABLE tmux_epochs (
  epoch      TEXT PRIMARY KEY,
  started_at INTEGER NOT NULL,
  seen_at    INTEGER NOT NULL
);

CREATE TABLE projects (
  id              TEXT PRIMARY KEY,          -- 沿用现有不可变 id，老链接不失效
  dir             TEXT NOT NULL UNIQUE,
  origin          TEXT NOT NULL DEFAULT '',  -- '' = discovered | 'user'
  display_name    TEXT,
  pinned          INTEGER NOT NULL DEFAULT 0,
  default_agent   TEXT,
  default_base    TEXT,
  first_seen      INTEGER NOT NULL,
  last_seen       INTEGER NOT NULL,
  last_session_at INTEGER NOT NULL DEFAULT 0,  -- 最后一次有会话的时刻
  archived_at     INTEGER                     -- 软退场，NULL = 在册
);
CREATE TABLE project_aliases (alias TEXT PRIMARY KEY, project_id TEXT NOT NULL);

-- 收尾留痕（merged / discarded / cleaned）。原先是 activity.log（JSONL，5MB 轮转一代）。
-- 并库时被漏下，代价是双重的：它与别的台账不在同一份快照里，备份/恢复各走各的；
-- 而后端一旦直连就没有文件路径了，于是每条留痕都被静默丢弃、项目活动恒空。
CREATE TABLE project_traces (
  id          TEXT PRIMARY KEY,
  repo        TEXT NOT NULL DEFAULT '',  -- = projects.dir
  branch      TEXT, head_oid TEXT, base TEXT,
  action      TEXT,                      -- merged | discarded | cleaned
  strategy    TEXT, merged_into TEXT, merged_kind TEXT,
  at          INTEGER NOT NULL DEFAULT 0
);
-- 文件形态靠轮转给读放大封顶；进了库改成按仓库各留最近 1000 条，写时修剪。
CREATE INDEX project_traces_repo ON project_traces(repo, at DESC);

-- 主键是**持久会话 id**（= tmux 会话名，见 docs/design/session-identity.md）：
-- 由创建时刻派生、不可变、跨 tmux server 重启唯一且永不复用 —— 天生就是对的主键。
CREATE TABLE sessions (
  id           TEXT PRIMARY KEY,
  label        TEXT,                      -- 展示名快照（会话死后 @roam_name 就没了）
  parent_id    TEXT REFERENCES sessions(id),
  project_id   TEXT REFERENCES projects(id),
  home_dir     TEXT,                      -- 归属目录（吸收 session-homes.json）
  worktree_dir TEXT,
  agent        TEXT,                      -- claude | codex | shell | …
  kind         TEXT,                      -- task | swarm-member | race | plugin | plain
  created_by   TEXT,
  created_at   INTEGER NOT NULL,
  spawn_cmd    TEXT,                      -- 重开所需的全部输入（§8）
  spawn_env    TEXT,                      -- JSON
  status       TEXT NOT NULL DEFAULT 'live',  -- live | dead
  died_at      INTEGER,
  died_reason   TEXT,                      -- killed | host-restart | unknown
  restored_from TEXT REFERENCES sessions(id),
  tmux_id      TEXT,                      -- 运行时句柄，只在 status=live 时有意义
  tmux_epoch   TEXT REFERENCES tmux_epochs(epoch)
);
CREATE INDEX sessions_project ON sessions(project_id, status);
CREATE UNIQUE INDEX sessions_handle ON sessions(tmux_epoch, tmux_id)
  WHERE tmux_epoch IS NOT NULL;
为什么主键必须换成持久会话 id。session-identity.md 已经把会话 id 定义成 id.ForSession(session_created, session_id) —— 它包含创建时刻，所以重启后新会话 拿到 $0 也撞不上旧会话。台账早就有一个跨重启唯一的键，只是没拿它当主键。换过来之后： 重启不再是「主键全部失效」，只是 tmux_id 置空；$N 复用天然无害， 因为 (epoch, tmux_id) 才是句柄；死会话的行可以安然留着，它的主键不依赖任何活着的东西。
